LL mulMod(ll a,ll b) {
    return ((a % MOD) * (b % MOD)) % MOD;
}
-----------------------------------------------------------------------------
LL addMod(LL a,LL b) {
    return ((a % MOD) + (b % MOD)) % MOD;
}
------------------------------------------------------------------------------
long long powMod(long long a,long long b){
    long long res = 1;
    while(b){
        if(b & 1) res = (res * a) % MOD;
        a = (a * a) % MOD;
        b >>= 1;
    }
    return res % MOD;
}
-------------------------------------------------------------------------------
long long modInverse(long long a, long long mod) {
    return powMod(a, mod - 2);
}
