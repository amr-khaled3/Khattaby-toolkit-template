using LL = long long;

vector<LL> getDivisors(LL n) {
    vector<LL> ret;
    LL i = 1;
    for (; i * i < n; i++) {
        if (n % i == 0) {
            ret.push_back(i);
            ret.push_back(n / i);
        }
    }
    if (i * i == n) ret.push_back(i);
    return ret;
}

vector<int> getMultiples(int x, int n) {
    vector<int> ret;
    for (int i = x; i <= n; i += x)
        ret.push_back(i);
    return ret;
}

vector<LL> divisorCounter() {
    const int MAX = 1e6 + 6;
    vector<LL> divisorCount(MAX + 1, 0);
    for (int i = 1; i <= MAX; ++i) {
        for (int j = i; j <= MAX; j += i) {
            divisorCount[j]++;
        }
    }
    return divisorCount;
}
