struct TECC { // 1 indexed
    int n, k;
    vector<vector<int>> g, t;
    vector<bool> used;
    vector<int> comp, ord, low;

    using edge = pair<int, int>;
    vector<edge> br;

    void dfs(int x, int prv, int &c) {
        used[x] = true;
        ord[x] = c++;
        low[x] = n + 1;

        bool mul = false;

        for (auto y : g[x]) {
            if (used[y]) {
                if (y != prv || mul)
                    low[x] = min(low[x], ord[y]);
                else
                    mul = true;
                continue;
            }

            dfs(y, x, c);
            low[x] = min(low[x], low[y]);
        }
    }

    void dfs2(int x, int num) {
        comp[x] = num;

        for (auto y : g[x]) {
            if (comp[y] != -1)
                continue;

            if (ord[x] < low[y]) {
                br.push_back({x, y});
                ++k;
                dfs2(y, k);
            } else {
                dfs2(y, num);
            }
        }
    }

    TECC(const vector<vector<int>> &graph)
        : g(graph),
          n((int)graph.size() - 1),
          used(n + 1, false),
          comp(n + 1, -1),
          ord(n + 1),
          low(n + 1),
          k(0) {

        int c = 0;

        for (int i = 1; i <= n; i++) {
            if (used[i])
                continue;

            dfs(i, -1, c);
            dfs2(i, k);
            ++k;
        }
    }

    void build_tree() {
        t.assign(k, {});

        for (auto [u, v] : br) {
            int x = comp[u];
            int y = comp[v];
            t[x].push_back(y);
            t[y].push_back(x);
        }
    }
};
