int n;
vector<vector<int>> adj;
vector<int> topo;
vector<bool> vis;

void dfs(int u) {
    vis[u] = true;

    for (int v : adj[u]) {
        if (!vis[v])
            dfs(v);
    }

    topo.push_back(u);
}

void topoSort() {
    topo.clear();
    vis.assign(n, false);

    for (int i = 0; i < n; i++) {
        if (!vis[i])
            dfs(i);
    }
    reverse(topo.begin(), topo.end());
}
