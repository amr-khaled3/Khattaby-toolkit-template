string multiplyStrings(string a, string b) {
    if (a == "0" || b == "0") return "0";

    int n = a.size(), m = b.size();
    vector<int> res(n + m, 0);

    for (int i = n - 1; i >= 0; i--) {
        for (int j = m - 1; j >= 0; j--) {
            int mul = (a[i] - '0') * (b[j] - '0');

            int sum = mul + res[i + j + 1];

            res[i + j + 1] = sum % 10;
            res[i + j] += sum / 10;
        }
    }

    string ans;

    for (int x : res) {
        if (!(ans.empty() && x == 0))
            ans += char(x + '0');
    }

    return ans.empty() ? "0" : ans;
}
