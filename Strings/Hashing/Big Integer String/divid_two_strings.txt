string divideStrings(string num, string divisor) {
    if (divisor == "0")
        return "Infinity";

    auto smaller = [](string a, string b) {
        if (a.size() != b.size())
            return a.size() < b.size();
        return a < b;
    };

    auto subtract = [](string a, string b) {
        string res;
        int carry = 0;

        reverse(a.begin(), a.end());
        reverse(b.begin(), b.end());

        for (int i = 0; i < a.size(); i++) {
            int x = a[i] - '0';
            int y = (i < b.size() ? b[i] - '0' : 0);

            x -= y + carry;

            if (x < 0) {
                x += 10;
                carry = 1;
            } else {
                carry = 0;
            }

            res += char(x + '0');
        }

        while (res.size() > 1 && res.back() == '0')
            res.pop_back();

        reverse(res.begin(), res.end());
        return res;
    };

    string cur, ans;

    for (char c : num) {
        cur += c;

        while (cur.size() > 1 && cur[0] == '0')
            cur.erase(cur.begin());

        int q = 0;

        while (!smaller(cur, divisor)) {
            cur = subtract(cur, divisor);
            q++;
        }

        ans += char(q + '0');
    }

    int pos = 0;
    while (pos < ans.size() - 1 && ans[pos] == '0')
        pos++;

    return ans.substr(pos);
}
